# S02 Tenancy (membership documents, roles, server-side company creation) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Company access is decided by a membership document with a role (Owner / Accountant / Viewer), enforced by Firestore rules and the server; companies are created only by the server (limit 3 owned per user); existing companies are migrated by a script.

**Architecture:** `organizations/{orgId}/members/{uid}` = `{uid, role, createdAt, addedBy}` replaces the `members` map on the org document. Rules use `hasRole(orgId, roles)` (one `exists`+`get` of the caller's member doc). The client lists its companies through a collection-group query on `members` filtered by its own `uid`. The server gets `requireOrgRole` and `POST /api/orgs` (Admin transaction: count owner memberships, create org + owner membership atomically). The UI reads its role from its own member doc and hides what the role cannot do (convenience only, the rules are the protection).

```ts
// Roles and what they may do (S02). Pure and import-free: used by the browser UI and by server code.
// This is a convenience layer for the UI; the real protection is firestore.rules and requireOrgRole.

export const ROLES = ['owner', 'accountant', 'viewer'] as const
export type Role = (typeof ROLES)[number]

export type Action =
  | 'data.write' // accounts, categories, transactions
  | 'bank.manage' // bank connections
  | 'company.edit' // company name and currency
  | 'subscription.view' // plan and billing

const ALLOWED: Record<Action, readonly Role[]> = {
  'data.write': ['owner', 'accountant'],
  'bank.manage': ['owner'],
  'company.edit': ['owner'],
  'subscription.view': ['owner'],
}

export function isRole(value: unknown): value is Role {
  return typeof value === 'string' && (ROLES as readonly string[]).includes(value)
}

/** `role` is null/undefined while it is unknown (loading) or when the user has no membership: nothing is allowed. */
export function can(role: Role | null | undefined, action: Action): boolean {
  return role != null && ALLOWED[action].includes(role)
}
```

```
rules_version = '2';

// FinFlow Firestore security rules (S02).
//
// Deny by default: any path that no `match` block below allows is closed to web/mobile
// clients. Server code (firebase-admin in lib/server) bypasses these rules entirely.
//
// Access is decided by a membership document:
//   organizations/{orgId}/members/{uid} = { uid, role: 'owner'|'accountant'|'viewer', createdAt, addedBy }
// Only the server writes organizations and memberships (POST /api/orgs, migration script).
//
// Data shape:
//   organizations/{orgId} { name, currency, ownerId, plan, trialEndsAt, createdAt }
//   organizations/{orgId}/{accounts|categories|transactions|bankConnections}/{docId}
//
// The name `members` is reserved for memberships: the collection-group rule below applies to
// every collection called `members`. New subcollections must be added to the lists below.
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function memberPath(orgId) {
      return /databases/$(database)/documents/organizations/$(orgId)/members/$(request.auth.uid);
    }

    // The caller has a membership document in the org and its role is one of `roles`.
    // exists() + get() of the same document is one read.
    function hasRole(orgId, roles) {
      return isSignedIn()
        && exists(memberPath(orgId))
        && get(memberPath(orgId)).data.role in roles;
    }

    // Type and size checks for the fields the owner may edit.
    function orgFieldsValid(o) {
      return o.name is string && o.name.size() > 0 && o.name.size() <= 120
        && o.currency is string && o.currency.size() == 3;
    }

    // A subcollection document may carry an `orgId`, but only its own org's.
    function orgIdMatchesPath(orgId) {
      return !('orgId' in request.resource.data) || request.resource.data.orgId == orgId;
    }

    match /organizations/{orgId} {
      // Any member may read the company. There is no `list` (companies are loaded by id from
      // the caller's memberships), no client `create` and no `delete`.
      allow get: if hasRole(orgId, ['owner', 'accountant', 'viewer']);

      // Allow-list: the owner may change only name and currency (settings page). Everything
      // else (ownerId, plan, trial, createdAt, any unknown field) is closed.
      allow update: if hasRole(orgId, ['owner'])
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name', 'currency'])
        && orgFieldsValid(request.resource.data);

      // Memberships are written by the server only.
      match /members/{memberId} {
        allow get: if isSignedIn() && (request.auth.uid == memberId || hasRole(orgId, ['owner']));
        allow list: if hasRole(orgId, ['owner']);
      }

      match /{subcollection}/{docId} {
        allow read: if (subcollection in ['accounts', 'categories', 'transactions']
            && hasRole(orgId, ['owner', 'accountant', 'viewer']))
          || (subcollection == 'bankConnections' && hasRole(orgId, ['owner']));

        allow create, update: if (subcollection in ['accounts', 'categories', 'transactions']
            && hasRole(orgId, ['owner', 'accountant'])
            && orgIdMatchesPath(orgId))
          || (subcollection == 'bankConnections'
            && hasRole(orgId, ['owner'])
            && orgIdMatchesPath(orgId));

        allow delete: if (subcollection in ['accounts', 'categories', 'transactions']
            && hasRole(orgId, ['owner', 'accountant']))
          || (subcollection == 'bankConnections' && hasRole(orgId, ['owner']));
      }
    }

    // "My companies": collectionGroup('members').where('uid', '==', <my uid>). A rule is not a
    // filter, so the query must constrain `uid` to the caller's own uid.
    match /{path=**}/members/{memberId} {
      allow read: if isSignedIn() && resource.data.uid == request.auth.uid;
    }

    // Server-only documents (e.g. _meta/health read by GET /api/health).
    match /_meta/{document=**} {
      allow read, write: if false;
    }
  }
}
```

```ts
/**
 * Returns the caller's role in organizations/{orgId} when it is one of `allowed`; otherwise
 * throws HttpError(403, 'forbidden') (same answer for "no such company", "not a member" and
 * "role not allowed", so ids cannot be probed). Reads organizations/{orgId}/members/{uid}
 * with the Admin SDK (bypasses security rules); a Firestore failure propagates (500).
 */
export async function requireOrgRole(uid: string, orgId: string, allowed: readonly Role[]): Promise<Role> {
  // orgId comes from the request: allow-list a single plain doc id, otherwise "org1/accounts/x"
  // would read a different document. The uid comes from a verified session but is checked too:
  // it becomes a path segment.
  if (typeof orgId !== 'string' || !ORG_ID.test(orgId) || RESERVED_ID.test(orgId)) {
    throw new HttpError(403, 'forbidden')
  }
  if (typeof uid !== 'string' || uid.length === 0 || uid.includes('/')) {
    throw new HttpError(403, 'forbidden')
  }

  const snap = await getAdmin().db.doc(`organizations/${orgId}/members/${uid}`).get()
  const role: unknown = snap.exists ? (snap.data() as { role?: unknown } | undefined)?.role : undefined
  if (!isRole(role) || !allowed.includes(role)) throw new HttpError(403, 'forbidden')
  return role
}
```

```ts
import 'server-only'
import { Timestamp } from 'firebase-admin/firestore'
import { getAdmin } from '@/lib/server/firebase-admin'
import { HttpError } from '@/lib/server/http-error'

export const MAX_OWNED_ORGS = 3
export const TRIAL_DAYS = 14
export const ORG_CURRENCIES = ['PLN', 'EUR', 'USD', 'UAH'] as const
export type OrgCurrency = (typeof ORG_CURRENCIES)[number]

const NAME_MAX = 120
const ALLOWED_KEYS = new Set(['name', 'currency'])
const DAY_MS = 24 * 60 * 60 * 1000

export interface CreateOrgInput {
  name: string
  currency: OrgCurrency
}

/** Validates the request body of POST /api/orgs. Throws HttpError(400) with a short code. */
export function parseCreateOrgInput(body: unknown): CreateOrgInput {
  if (typeof body !== 'object' || body === null || Array.isArray(body)) {
    throw new HttpError(400, 'invalid_body')
  }
  const record = body as Record<string, unknown>
  for (const key of Object.keys(record)) {
    if (!ALLOWED_KEYS.has(key)) throw new HttpError(400, 'invalid_body')
  }

  const name = typeof record.name === 'string' ? record.name.trim() : ''
  if (name.length < 1 || name.length > NAME_MAX) throw new HttpError(400, 'invalid_name')

  const currency = record.currency
  if (typeof currency !== 'string' || !(ORG_CURRENCIES as readonly string[]).includes(currency)) {
    throw new HttpError(400, 'invalid_currency')
  }
  return { name, currency: currency as OrgCurrency }
}

/**
 * Creates an organization and its owner membership in one Admin transaction after checking the
 * per-user limit. Returns the new organization id. The count and the writes are in the same
 * transaction, so two parallel requests cannot both slip past the limit.
 */
export async function createOrganizationForUser(uid: string, input: CreateOrgInput, now: Date = new Date()): Promise<string> {
  if (typeof uid !== 'string' || uid.length === 0 || uid.includes('/')) throw new HttpError(403, 'forbidden')

  const { db } = getAdmin()
  const orgRef = db.collection('organizations').doc()
  const memberRef = orgRef.collection('members').doc(uid)
  const createdAt = Timestamp.fromDate(now)
  const trialEndsAt = Timestamp.fromDate(new Date(now.getTime() + TRIAL_DAYS * DAY_MS))

  await db.runTransaction(async (tx) => {
    const memberships = await tx.get(db.collectionGroup('members').where('uid', '==', uid))
    const owned = memberships.docs.filter((d) => d.get('role') === 'owner').length
    if (owned >= MAX_OWNED_ORGS) throw new HttpError(409, 'org_limit_reached')

    tx.create(orgRef, {
      name: input.name,
      currency: input.currency,
      ownerId: uid,
      plan: 'trial',
      trialEndsAt,
      createdAt,
    })
    tx.create(memberRef, { uid, role: 'owner', createdAt, addedBy: uid })
  })

  return orgRef.id
}
```

```ts
import { requireUser } from '@/lib/server/auth'
import { errorResponse, HttpError } from '@/lib/server/http-error'
import { createOrganizationForUser, parseCreateOrgInput } from '@/lib/server/orgs'

export const runtime = 'nodejs'

// A legitimate body is `{"name":"<=120 chars","currency":"PLN"}`.
const MAX_BODY_BYTES = 2048

// POST /api/orgs  { name, currency }
// The only way to create a company: the browser never writes organizations or memberships
// (firestore.rules). Requires a session (and a same-origin request); a verified email is not required.
export async function POST(req: Request) {
  try {
    const { uid } = await requireUser(req)

    const declaredLength = req.headers.get('content-length')
    if (declaredLength !== null && Number(declaredLength) > MAX_BODY_BYTES) {
      throw new HttpError(413, 'payload too large')
    }
    if (!(req.headers.get('content-type') ?? '').toLowerCase().startsWith('application/json')) {
      throw new HttpError(415, 'unsupported media type')
    }
    const text = await req.text()
    if (text.length > MAX_BODY_BYTES) throw new HttpError(413, 'payload too large')

    let body: unknown
    try {
      body = JSON.parse(text)
    } catch {
      throw new HttpError(400, 'invalid_body')
    }

    const orgId = await createOrganizationForUser(uid, parseCreateOrgInput(body))
    return Response.json({ orgId }, { status: 201, headers: { 'Cache-Control': 'no-store' } })
  } catch (e) {
    return errorResponse(e)
  }
}
```

```ts
// Creates a company through POST /api/orgs (the browser never writes organizations: firestore.rules).
// The session cookie is sent automatically (same-origin).

export type OrgCreateErrorCode = 'org_limit_reached' | 'unauthorized' | 'invalid' | 'network' | 'server'

const MESSAGES: Record<OrgCreateErrorCode, string> = {
  org_limit_reached: 'You can own at most 3 companies.',
  unauthorized: 'Your session has expired. Please sign in again.',
  invalid: 'Please check the company name and currency.',
  network: 'Network error. Please check your connection and try again.',
  server: 'Could not create the company. Please try again.',
}

export class OrgCreateError extends Error {
  constructor(public readonly code: OrgCreateErrorCode) {
    super(MESSAGES[code])
    this.name = 'OrgCreateError'
  }
}

export async function createOrgViaApi(name: string, currency: string): Promise<string> {
  let res: Response
  try {
    res = await fetch('/api/orgs', {
      method: 'POST',
      credentials: 'same-origin',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, currency }),
    })
  } catch {
    throw new OrgCreateError('network')
  }

  const body = (await res.json().catch(() => null)) as { orgId?: unknown; error?: unknown } | null

  if (res.status === 201) {
    if (typeof body?.orgId === 'string' && body.orgId.length > 0) return body.orgId
    throw new OrgCreateError('server')
  }
  if (res.status === 409 && body?.error === 'org_limit_reached') throw new OrgCreateError('org_limit_reached')
  if (res.status === 401) throw new OrgCreateError('unauthorized')
  if (res.status === 400) throw new OrgCreateError('invalid')
  throw new OrgCreateError('server')
}
```

```ts
import { collectionGroup, doc, getDoc, getDocs, query, where } from 'firebase/firestore'
import { db } from './firebase'
import { getOrganization } from './firestore'
import { isRole, type Role } from '@/lib/shell/permissions'
import type { Organization } from '@/types'

export interface Membership {
  orgId: string
  role: Role
}

/** The role in a membership document, or null when the data is not a valid membership. */
export function roleFromMemberData(data: unknown): Role | null {
  if (typeof data !== 'object' || data === null) return null
  const role = (data as { role?: unknown }).role
  return isRole(role) ? role : null
}

/**
 * The caller's role in `orgId`, or null when there is no (valid) membership. Reading one's own
 * membership document is always allowed by the rules, so "no access" is an empty read, not an error.
 */
export async function getMembership(orgId: string, uid: string): Promise<Role | null> {
  const snap = await getDoc(doc(db, 'organizations', orgId, 'members', uid))
  return snap.exists() ? roleFromMemberData(snap.data()) : null
}

/**
 * The caller's memberships in every company. The query must filter `uid` to the caller's own
 * uid: the rules (match /{path=**}/members/{memberId}) are not a filter.
 */
export async function getMemberships(uid: string): Promise<Membership[]> {
  const snap = await getDocs(query(collectionGroup(db, 'members'), where('uid', '==', uid)))
  const result: Membership[] = []
  for (const d of snap.docs) {
    const role = roleFromMemberData(d.data())
    const orgId = d.ref.parent.parent?.id
    if (role && orgId) result.push({ orgId, role })
  }
  return result
}

/** The companies the user belongs to, oldest first (companies that cannot be loaded are skipped). */
export async function getMyOrganizations(uid: string): Promise<Organization[]> {
  const memberships = await getMemberships(uid)
  const orgs = await Promise.all(memberships.map((m) => getOrganization(m.orgId)))
  return orgs
    .filter((o): o is Organization => o !== null)
    .sort((a, b) => a.createdAt.getTime() - b.createdAt.getTime())
}
```

```ts
/** Reads a document as admin. Returns its raw Firestore `fields`, or null when it does not exist. */
export async function readDoc(
  request: APIRequestContext,
  path: string,
): Promise<Record<string, FirestoreValue> | null> {
  const url = `http://${firestoreHost()}/v1/projects/${PROJECT}/databases/(default)/documents/${path}`
  const res = await request.get(url, { headers: { Authorization: 'Bearer owner' } })
  if (res.status() === 404) return null
  if (!res.ok()) throw new Error(`read ${path} failed: ${res.status()} ${await res.text()}`)
  return ((await res.json()) as { fields?: Record<string, FirestoreValue> }).fields ?? {}
}

/** Creates organizations/{orgId}/members/{uid} as admin (the server's job in the real app). */
export async function seedMember(
  request: APIRequestContext,
  orgId: string,
  uid: string,
  role: 'owner' | 'accountant' | 'viewer',
  addedBy = uid,
): Promise<void> {
  await seedDoc(request, `organizations/${orgId}/members`, uid, { uid, role, createdAt: new Date(), addedBy })
}

/**
 * Writes a document through the Firestore REST API as a signed-in user (their ID token), so the
 * security rules apply. Returns the HTTP status: 200 allowed, 403 denied by the rules.
 */
export async function createDocAs(
  request: APIRequestContext,
  idToken: string,
  collectionPath: string,
  docId: string,
  data: SeedFields,
): Promise<number> {
  const url = `http://${firestoreHost()}/v1/projects/${PROJECT}/databases/(default)/documents/${collectionPath}?documentId=${encodeURIComponent(docId)}`
  const res = await request.post(url, {
    headers: { Authorization: `Bearer ${idToken}` },
    data: { fields: toFields(data) },
  })
  return res.status()
}
```

```js
// Pure planning for scripts/migrate-memberships.mjs (S02): which membership documents the legacy
// `organizations/{id}.members` maps imply. No I/O, no Firebase imports: unit-tested in Vitest.

const ROLES = ['owner', 'accountant', 'viewer']

/**
 * @param {Array<{ id: string, data: Record<string, unknown> }>} orgDocs
 * @returns {{ writes: Array<{orgId: string, uid: string, role: string, addedBy: string}>,
 *             skipped: Array<{orgId: string, uid: string | null, reason: string}> }}
 */
export function planMemberships(orgDocs) {
  const writes = []
  const skipped = []

  for (const { id: orgId, data } of orgDocs) {
    const members = data?.members
    if (typeof members !== 'object' || members === null || Array.isArray(members) || Object.keys(members).length === 0) {
      skipped.push({ orgId, uid: null, reason: 'no_members_map' })
      continue
    }
    const ownerId = typeof data.ownerId === 'string' && data.ownerId.length > 0 ? data.ownerId : null

    for (const uid of Object.keys(members)) {
      if (uid.length === 0 || uid.includes('/')) {
        skipped.push({ orgId, uid, reason: 'invalid_uid' })
        continue
      }
      const declared = members[uid]?.role
      const role = ROLES.includes(declared) ? declared : uid === ownerId ? 'owner' : null
      if (role === null) {
        skipped.push({ orgId, uid, reason: 'invalid_role' })
        continue
      }
      writes.push({ orgId, uid, role, addedBy: ownerId ?? uid })
    }
  }
  return { writes, skipped }
}
```

```js
#!/usr/bin/env node
// One-off migration (S02): legacy `organizations/{id}.members` map -> membership documents
// `organizations/{id}/members/{uid}` = { uid, role, createdAt, addedBy }.
//
//   node scripts/migrate-memberships.mjs                       dry run: prints counts and org ids, writes nothing
//   node scripts/migrate-memberships.mjs --apply               creates the missing membership documents (idempotent)
//   node scripts/migrate-memberships.mjs --drop-members-map    removes the legacy `members` field, only after every
//                                                              membership document is verified (run AFTER the
//                                                              rollout, see the S02 spec)
//
// Credentials (never printed): GOOGLE_APPLICATION_CREDENTIALS, else the service-account JSON in .secrets/
// (finflow-*-firebase-adminsdk-*.json). Nothing from document contents is printed, only counters and ids.
import fs from 'node:fs'
import path from 'node:path'
import { initializeApp, cert, applicationDefault } from 'firebase-admin/app'
import { getFirestore, FieldValue } from 'firebase-admin/firestore'
import { planMemberships } from './lib/plan-memberships.mjs'

const PROJECT_ID = 'finflow-e94b5'
const args = new Set(process.argv.slice(2))
const APPLY = args.has('--apply')
const DROP = args.has('--drop-members-map')
const CHUNK = 400

if (APPLY && DROP) {
  console.error('Use --apply and --drop-members-map in separate runs.')
  process.exit(2)
}

function credential() {
  if (process.env.GOOGLE_APPLICATION_CREDENTIALS) return applicationDefault()
  const dir = path.resolve('.secrets')
  const file = fs.existsSync(dir) ? fs.readdirSync(dir).find((f) => /^finflow-.*-firebase-adminsdk-.*\.json$/.test(f)) : undefined
  if (!file) {
    console.error('No credentials: set GOOGLE_APPLICATION_CREDENTIALS or put the service-account JSON in .secrets/.')
    process.exit(2)
  }
  return cert(JSON.parse(fs.readFileSync(path.join(dir, file), 'utf8')))
}

initializeApp({ credential: credential(), projectId: PROJECT_ID })
const db = getFirestore()

const orgSnap = await db.collection('organizations').get()
const orgDocs = orgSnap.docs.map((d) => ({ id: d.id, data: d.data() }))
const plan = planMemberships(orgDocs)

const existing = new Map() // `${orgId}/${uid}` -> role | null
for (let i = 0; i < plan.writes.length; i += CHUNK) {
  const refs = plan.writes.slice(i, i + CHUNK).map((w) => db.doc(`organizations/${w.orgId}/members/${w.uid}`))
  const snaps = await db.getAll(...refs)
  snaps.forEach((s, j) => {
    const w = plan.writes[i + j]
    existing.set(`${w.orgId}/${w.uid}`, s.exists ? (s.data().role ?? null) : undefined)
  })
}
const toCreate = plan.writes.filter((w) => existing.get(`${w.orgId}/${w.uid}`) === undefined)
const alreadyThere = plan.writes.filter((w) => existing.get(`${w.orgId}/${w.uid}`) === w.role)
const conflicts = plan.writes.filter((w) => {
  const e = existing.get(`${w.orgId}/${w.uid}`)
  return e !== undefined && e !== w.role
})

console.log(`organizations: ${orgDocs.length}`)
console.log(`membership documents implied by legacy maps: ${plan.writes.length}`)
console.log(`  to create: ${toCreate.length}, already present: ${alreadyThere.length}, role conflicts (left untouched): ${conflicts.length}`)
console.log(`skipped: ${plan.skipped.length}`)
for (const s of plan.skipped) console.log(`  skipped org=${s.orgId} uid=${s.uid ?? '-'} reason=${s.reason}`)
for (const c of conflicts) console.log(`  conflict org=${c.orgId} uid=${c.uid}`)

if (DROP) {
  // Safety: refuse unless every membership implied by a legacy map exists with the planned role.
  const missing = plan.writes.filter((w) => existing.get(`${w.orgId}/${w.uid}`) !== w.role)
  if (missing.length > 0) {
    console.error(`REFUSED: ${missing.length} membership document(s) are missing or differ; run --apply first and verify.`)
    for (const m of missing) console.error(`  org=${m.orgId} uid=${m.uid}`)
    process.exit(1)
  }
  const withMap = orgDocs.filter((o) => Object.hasOwn(o.data, 'members'))
  for (let i = 0; i < withMap.length; i += CHUNK) {
    const batch = db.batch()
    for (const o of withMap.slice(i, i + CHUNK)) batch.update(db.doc(`organizations/${o.id}`), { members: FieldValue.delete() })
    await batch.commit()
  }
  console.log(`removed the legacy members field from ${withMap.length} organization(s)`)
  process.exit(0)
}

if (!APPLY) {
  console.log('DRY RUN: nothing written. Re-run with --apply to create the missing membership documents.')
  process.exit(0)
}

const orgById = new Map(orgDocs.map((o) => [o.id, o.data]))
for (let i = 0; i < toCreate.length; i += CHUNK) {
  const batch = db.batch()
  for (const w of toCreate.slice(i, i + CHUNK)) {
    const created = orgById.get(w.orgId)?.createdAt
    batch.create(db.doc(`organizations/${w.orgId}/members/${w.uid}`), {
      uid: w.uid,
      role: w.role,
      createdAt: created && typeof created.toDate === 'function' ? created : FieldValue.serverTimestamp(),
      addedBy: w.addedBy,
    })
  }
  await batch.commit()
}

// Verify by re-reading.
let bad = 0
for (let i = 0; i < plan.writes.length; i += CHUNK) {
  const slice = plan.writes.slice(i, i + CHUNK)
  const snaps = await db.getAll(...slice.map((w) => db.doc(`organizations/${w.orgId}/members/${w.uid}`)))
  snaps.forEach((s, j) => {
    const w = slice[j]
    if (!s.exists || s.data().role !== w.role || s.data().uid !== w.uid) bad++
  })
}
console.log(`created: ${toCreate.length}; verification mismatches: ${bad}`)
process.exit(bad === 0 ? 0 : 1)
```
